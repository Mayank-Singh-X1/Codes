1192. Critical Connections in a Network
class Solution {
    void dfs(int node, int parent,vector<int>& disc, vector<int>& low,vector<bool>& visited,unordered_map<int, list<int>>& graph,vector<vector<int>>& bridges,int& timer) {

        visited[node] = true;
        disc[node] = low[node] = timer++;

        for (int nbr : graph[node]) {

            if (nbr == parent) continue;

            // Tree edge
            if (!visited[nbr]) {
                dfs(nbr, node, disc, low, visited, graph, bridges, timer);

                // Update low value after returning from child
                low[node] = min(low[node], low[nbr]);

                // 🔑 BRIDGE CONDITION
                if (low[nbr] > disc[node]) {
                    bridges.push_back({node, nbr});
                }
            }
            // Back edge
            else {
                low[node] = min(low[node], disc[nbr]);
            }
        }
    }
public:
    vector<vector<int>> criticalConnections(int n, vector<vector<int>>& connections) {
         unordered_map<int, list<int>> graph;
        for (auto& e : connections) {
            graph[e[0]].push_back(e[1]);
            graph[e[1]].push_back(e[0]);
        }

        vector<int> disc(n, -1), low(n, -1);
        vector<bool> visited(n, false);
        vector<vector<int>> bridges;

        int timer = 0;

        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                dfs(i, -1, disc, low, visited, graph, bridges, timer);
            }
        }

        return bridges;
    }
};
